function(wing_test name)
    add_executable(${name} ${ARGN})
    target_include_directories(${name} PRIVATE ${USER}/Inc stubs)
    target_compile_definitions(${name} PRIVATE TOP_WING_HOST_TEST TOP_WING_CAN_BUS=2)
    target_compile_options(${name} PRIVATE -Wall -Wextra -UNDEBUG -ffunction-sections -fdata-sections)
    target_link_options(${name} PRIVATE -Wl,--gc-sections)
    target_link_libraries(${name} PRIVATE abox::mqtt_digest abox::mqtt_runtime abox::cloud_service abox::boot_v2_app)
    add_test(NAME ${name} COMMAND ${name})
endfunction()
